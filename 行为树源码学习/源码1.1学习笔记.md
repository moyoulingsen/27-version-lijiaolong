## 源码1.1学习笔记

```cpp
// Example of custom SyncActionNode (synchronous action)
// without ports.
class ApproachObject : public BT::SyncActionNode
                     //public BT::SyncActionNode：继承 BehaviorTree.CPP 提供的同步动作节点基类。
{
public:
  ApproachObject(const std::string& name) :
                 //也就是行为树中该节点的实例名。
      BT::SyncActionNode(name, {})
    //BT::SyncActionNode(name, {}) 调用基类构造函数。
                             //同BT::PortsList{}
    //基类构造函数SyncActionNode(const std::string& name, const PortsList& ports);
  {}

  // You must override the virtual function tick()
  BT::NodeStatus tick() override
//BT::NodeStatus 是 BehaviorTree.CPP 库中定义的一个枚举类型，用来表示行为树节点执行后的状态。
/*{
  IDLE,      // 空闲，尚未执行
  RUNNING,   // 正在执行，还没结束
  SUCCESS,   // 执行成功
  FAILURE,   // 执行失败
  SKIPPED    // 被跳过（某些版本有）
};*/
      
  {
    std::cout << "ApproachObject: " << this->name() << std::endl;
    return BT::NodeStatus::SUCCESS;
  }//就像这样
};
```
